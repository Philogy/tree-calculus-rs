use logos::{Lexer, Logos};
use std::{collections::HashMap, ops::Range, rc::Rc};

struct Trees {
    names: HashMap<Box<str>, Rc<Tree>>,
}

impl Trees {
    fn new() -> Self {
        Self {
            names: HashMap::new(),
        }
    }
}

type Span = Range<usize>;


fn parse_paren_tree<'src>(
    lexer: &mut LexerWrapper<'src>,
    trees: &mut Trees,
) -> Result<Rc<Tree>, Error> {
    let (Token::Open, _) = lexer.next() else {
        unreachable!("(tree) not started with '('")
    };
    let tree = parse_tree(lexer, trees, true)?.expect("root=true must return some or error");
    match lexer.next() {
        (Token::Close, _) => {}
        (tok, span) => {
            return Err((format!("Unexpected token {:?}, expected ')'", tok), span));
        }
    }
    Ok(tree)
}

fn parse_tree_ref<'src>(
    lexer: &mut LexerWrapper<'src>,
    trees: &mut Trees,
) -> Result<Rc<Tree>, Error> {
    let (Token::Identifier(name), span) = lexer.next() else {
        unreachable!("tree ref missing identifier")
    };
    match trees.names.get(name) {
        Some(t) => Ok(t.clone()),
        None => Err((format!("Undefined tree {:?}", name), span)),
    }
}

fn parse_tree<'src>(
    lexer: &mut LexerWrapper<'src>,
    trees: &mut Trees,
    root: bool,
) -> Result<Option<Rc<Tree>>, Error> {
    let mut t = None;
    let mut iters = 0;
    while let Some(next_tree_to_apply) = match lexer.peek() {
        (Token::Identifier(_), _) => Some(parse_tree_ref(lexer, trees)?),
        (Token::Triangle, _) if root => {
            lexer.next();
            let t = if let Some(lhs) = parse_tree(lexer, trees, false)? {
                if let Some(rhs) = parse_tree(lexer, trees, false)? {
                    Rc::new(Tree::Fork(lhs, rhs))
                } else {
                    Rc::new(Tree::Stem(lhs))
                }
            } else {
                Rc::new(Tree::Leaf)
            };
            Some(t)
        }
        (Token::Triangle, _) if !root => {
            lexer.next();
            Some(Rc::new(Tree::Leaf))
        }
        (Token::Open, _) => Some(parse_paren_tree(lexer, trees)?),
        (Token::Eof | Token::Newline | Token::Close, _) if !root || t.is_some() => {
            // Potentially closing different expr or valid Eof
            None
        }
        (tok, span) => {
            return Err((
                format!("Unexpected token {:?}, expected △ / (tree) / name", tok),
                span,
            ));
        }
    } {
        if !root {
            return Ok(Some(next_tree_to_apply));
        }

        iters += 1;
        if iters >= 100 {
            return Err((format!("Infinite parse"), lexer.lexer.span()));
        }
        t = Some(match t {
            None => next_tree_to_apply,
            Some(prev_tree) => rec_apply(&prev_tree, &next_tree_to_apply),
        });
    }
    Ok(t)
}

fn parse_decl<'src>(
    lexer: &mut LexerWrapper<'src>,
    trees: &mut Trees,
) -> Result<(&'src str, Rc<Tree>, Span), Error> {
    let (Token::Identifier(name), span) = lexer.next() else {
        unreachable!("Expected decl to start with ident");
    };

    match lexer.next() {
        (Token::Equal, _) => {}
        (tok, span) => return Err((format!("Unexpected token {:?}, expected =", tok), span)),
    }

    let tree = parse_tree(lexer, trees, true)?.expect("root=true, must return or error");

    Ok((name, tree, span))
}

fn parse<'src>(mut lexer: LexerWrapper<'src>) -> Result<Trees, Error> {
    let mut trees = Trees::new();

    while let (Token::Identifier(_), _) = lexer.peek_skip_newline() {
        let (new_name, tree, span) = parse_decl(&mut lexer, &mut trees)?;
        if trees.names.insert(new_name.into(), tree).is_some() {
            return Err((format!("Duplicate definition of {:?}", new_name), span));
        }
    }

    match lexer.next() {
        (Token::Eof, _) => {}
        (tok, span) => return Err((format!("Unexpected token {:?}, expected EOF", tok), span)),
    }

    Ok(trees)
}

fn pretty_err<'src>(source: &'src str, reason: String, span: Span, with_line_number: bool) {
    let newlines: Vec<_> = source
        .char_indices()
        .filter_map(|(i, c)| (c == '\n').then_some(i))
        .collect();
    use std::iter::once;
    for (line, (start, end)) in once(0)
        .chain(newlines.iter().copied())
        .zip(newlines.iter().copied().chain(once(source.len())))
        .enumerate()
    {
        let line = line + 1;
        if start <= span.start && span.start < end {
            let start = if line > 1 { start + 1 } else { start };
            let size = if with_line_number {
                eprintln!("{} | {}", line, &source[start..end]);
                4 + line.checked_ilog10().unwrap_or(0)
            } else {
                eprintln!("{}", &source[start..end]);
                0
            };
            for _ in 0..size {
                eprint!(" ");
            }
            for j in start..end {
                if j < span.start {
                    eprint!(" ");
                } else if j < span.end {
                    eprint!("^")
                }
            }
        }
    }
    eprintln!("");
    eprintln!("{}", reason);
}

fn tree_repl(mut trees: Trees) {
    let mut input = String::with_capacity(4096);
    use std::io::{self, Write};
    loop {
        input.clear();
        print!("\n> "); // prompt
        io::stdout().flush().unwrap(); // ensure the prompt is shown

        match io::stdin().read_line(&mut input) {
            Ok(0) => {
                println!("Exiting...");
                break;
            }
            Ok(_) => {
                let input = input.trim();
                if input == "/exit" || input == "/quit" {
                    println!("Exiting...");
                    break;
                }
                if let Some(name) = input.strip_prefix("/show ") {
                    match trees.names.get(name) {
                        Some(tree) => println!("{:?} = {:?}", name, tree),
                        None => eprintln!("{:?} undefined", name),
                    }
                    continue;
                }

                let lexer = if input.contains('=') {
                    let mut lexer = LexerWrapper::new(input);
                    match parse_decl(&mut lexer, &mut trees) {
                        Ok((new_name, tree, _)) => {
                            println!("=> {}", &tree);
                            trees.names.insert(new_name.into(), tree);
                            Some(lexer)
                        }
                        Err((reason, span)) => {
                            pretty_err(input, reason, span, false);
                            None
                        }
                    }
                } else {
                    let mut lexer = LexerWrapper::new(input);
                    match parse_tree(&mut lexer, &mut trees, true) {
                        Ok(Some(tree)) => {
                            println!("=> {}", &tree);
                            Some(lexer)
                        }
                        Ok(None) => unreachable!("reachable?"),
                        Err((reason, span)) => {
                            pretty_err(input, reason, span, false);
                            None
                        }
                    }
                };
                if let Some(mut lexer) = lexer {
                    let (tok, span) = lexer.next_skip_newline();
                    if !matches!(tok, Token::Eof) {
                        pretty_err(
                            input,
                            format!("Unexpected token <{:?}>, expected end of input", tok),
                            span,
                            false,
                        );
                    }
                }
            }
            Err(e) => {
                eprintln!("Error reading input: {}", e);
                break;
            }
        }
    }
}

fn main() {
    let args: Vec<_> = std::env::args().collect();

    let source = std::fs::read_to_string(
        args.get(1)
            .expect("Missing input file path, expected: [FILE] [PROG_NAME] [INPUT_NAME]"),
    )
    .unwrap();

    let trees = match parse(LexerWrapper::new(&source)) {
        Ok(trees) => trees,
        Err((reason, span)) => {
            pretty_err(&source, reason, span, true);
            std::process::exit(1);
        }
    };

    tree_repl(trees);
}
